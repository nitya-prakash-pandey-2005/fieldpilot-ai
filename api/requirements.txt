# FieldPilot AI — API gateway runtime dependencies.
#
# Everything here is imported UNGUARDED somewhere on a live request path, so a
# missing entry is a 500 at demo time rather than a degraded feature. Optional
# extras (depth model, OCR, document AI, training) live in the sections below —
# install those explicitly.
#
#   pip install -r api/requirements.txt              # runs the API
#   pip install -r api/requirements.txt -r api/requirements-optional.txt

# --- web layer ---
fastapi
uvicorn[standard]
pydantic>=2.0
python-multipart      # FastAPI UploadFile / Form parsing — required by every upload route
sse-starlette
websockets
httpx                 # real Slack webhook POSTs (agents/notification/router.py)
requests

qdrant-client

supervision
onnx

# --- orchestration (the 10-agent graph) ---
# agents/orchestrator/graph.py imports these at module scope, and api/main.py
# imports routes.orchestrator at startup — so without langgraph the API does not
# boot at all, it does not merely lose the graph. Was missing from every
# requirements file until an audit caught it against a clean interpreter.
langgraph             # StateGraph/START/END + MemorySaver checkpointer

# --- language / voice ---
groq                  # Whisper STT + LLM (agents/voice/transcriber.py) — unguarded import
sentence-transformers # BGE embeddings for RAG (agents/memory/retriever.py)
